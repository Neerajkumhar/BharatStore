PHASE 1 - DATABASE SCHEMA (New Models)
==============================================================================

File: packages/database/prisma/schema.prisma

--- Model 1: SubscriptionPlan ---
  Defines the pricing tiers available for businesses.

  model SubscriptionPlan {
    id              String   @id @default(uuid()) @db.Uuid
    name            String   @db.VarChar(50)       // "Free", "Starter", "Pro", "Enterprise"
    slug            String   @unique @db.VarChar(50)
    description     String?
    monthlyPrice    Decimal  @db.Decimal(10, 2)    // INR per month
    annualPrice     Decimal  @db.Decimal(10, 2)    // INR per year
    currency        String   @default("INR") @db.VarChar(3)
    maxProducts     Int      @default(50)
    maxOrders       Int      @default(500)
    maxStaff        Int      @default(3)
    maxStorageMb    Int      @default(500)          // MB
    features        String[] @default([])           // feature flag slugs enabled
    isActive        Boolean  @default(true)
    displayOrder    Int      @default(0)
    createdAt       DateTime @default(now())
    updatedAt       DateTime @updatedAt

    subscriptions   TenantSubscription[]
  }

--- Model 2: TenantSubscription ---
  Links a tenant to their active plan.

  model TenantSubscription {
    id              String   @id @default(uuid()) @db.Uuid
    tenantId        String   @unique @db.Uuid
    planId          String   @map("plan_id") @db.Uuid
    status          SubscriptionStatus @default(TRIAL)
    trialEndsAt     DateTime? @map("trial_ends_at")
    currentPeriodStart DateTime? @map("current_period_start")
    currentPeriodEnd   DateTime? @map("current_period_end")
    cancelAt        DateTime? @map("cancel_at")
    cancelledAt     DateTime? @map("cancelled_at")
    paymentMethod   String?  @map("payment_method") @db.VarChar(50)
    razorpaySubId   String?  @map("razorpay_sub_id") @db.VarChar(100)
    notes           String?
    createdAt       DateTime @default(now())
    updatedAt       DateTime @updatedAt

    tenant          Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
    plan            SubscriptionPlan @relation(fields: [planId], references: [id], onDelete: Restrict)
  }

  enum SubscriptionStatus {
    TRIAL
    ACTIVE
    PAST_DUE
    CANCELLED
    SUSPENDED
    EXPIRED
  }

--- Model 3: FeatureFlag ---
  Platform-wide feature toggles that can be enabled per plan or per tenant.

  model FeatureFlag {
    id              String   @id @default(uuid()) @db.Uuid
    slug            String   @unique @db.VarChar(100) // "pos_counter", "gst_invoicing", "whatsapp_broadcast"
    name            String   @db.VarChar(100)
    description     String?
    category        String   @db.VarChar(50)          // "commerce", "finance", "marketing", "analytics"
    isPlatformWide  Boolean  @default(false)          // true = enabled for ALL tenants
    createdAt       DateTime @default(now())

    planFeatures    PlanFeature[]
    tenantOverrides TenantFeatureOverride[]
  }

--- Model 4: PlanFeature (junction) ---
  Which features each plan unlocks.

  model PlanFeature {
    planId          String @map("plan_id") @db.Uuid
    featureId       String @map("feature_id") @db.Uuid

    plan            SubscriptionPlan @relation(fields: [planId], references: [id], onDelete: Cascade)
    feature         FeatureFlag      @relation(fields: [featureId], references: [id], onDelete: Cascade)

    @@id([planId, featureId])
  }

--- Model 5: TenantFeatureOverride ---
  Per-tenant feature overrides (grant/revoke specific features).

  model TenantFeatureOverride {
    id              String   @id @default(uuid()) @db.Uuid
    tenantId        String   @map("tenant_id") @db.Uuid
    featureId       String   @map("feature_id") @db.Uuid
    enabled         Boolean  @default(true)
    grantedById     String?  @map("granted_by_id") @db.Uuid
    reason          String?
    expiresAt       DateTime?
    createdAt       DateTime @default(now())

    tenant          Tenant   @relation(fields: [tenantId], references: [id], onDelete: Cascade)
    feature         FeatureFlag @relation(fields: [featureId], references: [id], onDelete: Cascade)

    @@unique([tenantId, featureId])
  }

--- Model 6: PlatformAuditLog ---
  Super admin actions logged separately from tenant audit logs.

  model PlatformAuditLog {
    id              String   @id @default(uuid()) @db.Uuid
    actorId         String   @map("actor_id") @db.Uuid
    actorEmail      String   @map("actor_email")
    action          String   @db.VarChar(100)     // "tenant.suspend", "plan.change", "feature.grant"
    targetType      String   @map("target_type") @db.VarChar(50) // "tenant", "user", "plan", "feature"
    targetId        String?  @map("target_id") @db.Uuid
    targetName      String?  @map("target_name")
    beforeState     Json?    @map("before_state")
    afterState      Json?    @map("after_state")
    ipAddress       String?  @map("ip_address")
    createdAt       DateTime @default(now())

    @@index([actorId])
    @@index([targetType, targetId])
    @@index([createdAt])
    @@map("platform_audit_logs")
  }

--- Model 7: TenantUsage ---
  Track monthly usage per tenant for limits enforcement.

  model TenantUsage {
    id              String   @id @default(uuid()) @db.Uuid
    tenantId        String   @map("tenant_id") @db.Uuid
    period          String   @db.VarChar(7)       // "2026-09"
    productsCount   Int      @default(0) @map("products_count")
    ordersCount     Int      @default(0) @map("orders_count")
    staffCount      Int      @default(0) @map("staff_count")
    storageUsedMb   Decimal  @default(0) @map("storage_used_mb") @db.Decimal(10, 2)
    apiCalls        Int      @default(0) @map("api_calls")
    createdAt       DateTime @default(now())
    updatedAt       DateTime @updatedAt

    @@unique([tenantId, period])
    @@map("tenant_usage")
  }

    1. Extract session token from cookie/header
    2. Verify JWT
    3. Check session.isSuperAdmin === true

PHASE 4 - SUPER ADMIN API ROUTES
==============================================================================

All under apps/web/app/api/superadmin/

--- 4a. Auth & Session ---

  POST   /api/superadmin/logout         # Clear session

--- 4b. Platform Overview ---
  GET    /api/superadmin/overview       # Dashboard stats (counts, revenue, growth)
  GET    /api/superadmin/analytics      # Charts data (tenant growth, revenue trends)

--- 4c. Tenants Management ---
  GET    /api/superadmin/tenants        # List all tenants (paginated, filterable)
  GET    /api/superadmin/tenants/[id]   # Tenant detail (owner, subscription, usage, stats)
  PUT    /api/superadmin/tenants/[id]   # Update tenant (legal name, status, etc.)
  DELETE /api/superadmin/tenants/[id]   # Soft delete (deactivate tenant)
  POST   /api/superadmin/tenants/[id]/suspend   # Suspend tenant
  POST   /api/superadmin/tenants/[id]/reactivate # Reactivate tenant
  POST   /api/superadmin/tenants/[id]/impersonate # Generate impersonation session

--- 4d. Owners Management ---
  GET    /api/superadmin/owners         # List all owners
  GET    /api/superadmin/owners/[id]    # Owner detail (all tenants, activity)
  PUT    /api/superadmin/owners/[id]    # Update owner (name, email, status)
  POST   /api/superadmin/owners/[id]/suspend  # Suspend owner (all tenants)
  POST   /api/superadmin/owners/[id]/reactivate

--- 4e. Subscription Plans ---
  GET    /api/superadmin/plans          # List all plans
  POST   /api/superadmin/plans          # Create plan
  GET    /api/superadmin/plans/[id]     # Plan detail with features
  PUT    /api/superadmin/plans/[id]     # Update plan
  DELETE /api/superadmin/plans/[id]     # Archive plan

--- 4f. Tenant Subscriptions ---
  GET    /api/superadmin/subscriptions               # List all subscriptions
  PUT    /api/superadmin/subscriptions/[tenantId]    # Change tenant's plan
  POST   /api/superadmin/subscriptions/[tenantId]/trial/extend  # Extend trial
  POST   /api/superadmin/subscriptions/[tenantId]/cancel        # Cancel subscription

--- 4g. Feature Flags ---
  GET    /api/superadmin/features       # List all feature flags
  POST   /api/superadmin/features       # Create feature flag
  PUT    /api/superadmin/features/[id]  # Update feature flag
  DELETE /api/superadmin/features/[id]  # Delete feature flag
  POST   /api/superadmin/features/[id]/toggle  # Toggle platform-wide

--- 4h. Tenant Feature Overrides ---
  GET    /api/superadmin/tenants/[id]/features   # List tenant's features
  POST   /api/superadmin/tenants/[id]/features   # Grant feature to tenant
  DELETE /api/superadmin/tenants/[id]/features/[featureId]  # Revoke feature

--- 4i. Platform Audit ---
  GET    /api/superadmin/audit          # Platform audit logs (paginated)

--- 4j. Platform Health ---
  GET    /api/superadmin/health         # Platform health metrics
  GET    /api/superadmin/usage          # All tenant usage data

PHASE 6 - TENANT LIMITS ENFORCEMENT
==============================================================================

--- 6a. Middleware check (middleware.ts or new lib/limits.ts)

  Before tenant API operations, check:
    1. Is tenant subscription ACTIVE? (not suspended/cancelled)
    2. Does tenant have feature enabled?
    3. Is tenant within usage limits?

  Enforcement points:
    - POST /api/products → check productsCount < plan.maxProducts
    - POST /api/orders → check ordersCount < plan.maxOrders
    - POST /api/admin/staff → check staffCount < plan.maxStaff
    - POST /api/upload → check storageUsedMb < plan.maxStorageMb

--- 6b. Usage tracking (api/usage/route.ts)

  On each relevant API call, increment usage counters:
    - Product create/delete → update productsCount
    - Order create → update ordersCount
    - Staff invite/accept → update staffCount

  Use upsert with atomic increments for the current month period.

--- 6c. Upgrade prompts

  When limit reached:
    - Return 403 with specific error:
      { error: "Plan limit reached", code: "PLAN_LIMIT_EXCEEDED",
        resource: "products", current: 50, limit: 50, plan: "free",
        upgradeUrl: "/superadmin/tenants/[id]?tab=subscription" }
    - Frontend shows upgrade prompt modal

PHASE 7 - IMPERSONATION (Super Admin Login-as-Owner)
==============================================================================

--- 7a. Impersonation API (api/superadmin/tenants/[id]/impersonate)

  POST /api/superadmin/tenants/[id]/impersonate

  Flow:
    1. Super admin clicks "Impersonate" on tenant detail page
    2. API generates a short-lived JWT (15 min) with:
       - userId: target tenant's owner userId
       - tenantId: target tenant id
       - impersonatedBy: super admin userId
       - isImpersonation: true
    3. Returns impersonation token
    4. Frontend sets cookie and redirects to /dashboard
    5. Dashboard shows "Impersonating [Business Name]" banner with "Exit" button
    6. Audit log entry created

--- 7b. Impersonation detection (lib/auth.ts)

  In JWT payload:
    isImpersonation?: boolean
    impersonatedBy?: string

  If isImpersonation is true:
    - Show impersonation banner in dashboard layout
    - Log all actions with impersonator context
    - Session expires in 15 minutes
    - Exit impersonation clears cookie and redirects to /superadmin
